   - Load actor from DB, validate exists (404 if not)
   - If `actor.lora_training_status` is None → return `{"status": null, "lora_url": null, "trained_at": null}`
   - If status is terminal (COMPLETED or FAILED) → return current DB values directly (no Replicate poll)
   - If status is non-terminal (QUEUED or TRAINING):
     - Load UserSettings to get replicate_api_token
     - If no token, return current DB values (can't poll without token)
     - Create `ReplicateBackend` and call `poll_status(actor.lora_training_job_id)`
     - If status changed to COMPLETED:
       - Call `download_and_store_weights(job.weights_url, str(actor.id), storage)` to download and store the .safetensors

     - If status changed to FAILED:
       - Update actor: `lora_training_status = "FAILED"`
     - Otherwise: update `lora_training_status` with current status
     - Commit
   - Return `{"status": actor.lora_training_status, "lora_url": actor.lora_url, "trained_at": actor.lora_trained_at.isoformat() if actor.lora_trained_at else None, "error": job.error if applicable else None}`

   export async function trainActorLora(actorId: string): Promise<TrainLoraResponse> {
     const res = await fetch(`${BASE}/asset-library/actors/${actorId}/train-lora`, { method: 'POST' });
     if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.detail || 'Failed to start training'); }
     return res.json();
   }

   export async function getActorLoraStatus(actorId: string): Promise<LoraStatusResponse> {
     const res = await fetch(`${BASE}/asset-library/actors/${actorId}/lora-status`);
     if (!res.ok) throw new Error('Failed to fetch LoRA status');
     return res.json();
   }
