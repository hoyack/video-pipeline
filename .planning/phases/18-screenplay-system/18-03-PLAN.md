```typescript
export interface CharacterBreakdownEntry {

  role: string;
  arc: string;
  description: string;
  manifest_tag: string | null;
}

/** Scene breakdown entry within a screenplay */
export interface SceneBreakdownEntry {
  scene_number: number;
  slugline: string;
  intent: string;
  emotional_beat: string;
  story_state_in: string;
  story_state_out: string;
  characters_present: string[];
  set_ref: string | null;
  props_required: string[];
}

/** Shot list entry within a screenplay */
export interface ShotListEntry {
  scene_number: number;
  shot_number: number;
  shot_type: string;
  camera_movement: string;
  action_notes: string;
  duration_hint: string;
}

/** Response from GET /api/productions/{id}/screenplay */
export interface ScreenplayResponse {
  id: string;
  production_id: string;
  title: string | null;
  genre: string | null;
  status: string; // DRAFT | IN_REVIEW | LOCKED
  logline: string | null;
  treatment: string | null;
  character_breakdowns: CharacterBreakdownEntry[] | null;
  scene_breakdown: SceneBreakdownEntry[] | null;
  script: string | null;
  shot_list: ShotListEntry[] | null;
  text_model: string | null;
  generating_step: string | null;
  created_at: string;
  updated_at: string;
}

/** Request body for PUT /api/productions/{id}/screenplay */
export interface ScreenplayUpdate {
  title?: string;
  genre?: string;
  logline?: string;
  treatment?: string;
  character_breakdowns?: CharacterBreakdownEntry[];
  scene_breakdown?: SceneBreakdownEntry[];
  script?: string;
  shot_list?: ShotListEntry[];
  text_model?: string;
}

/** Scene created from screenplay breakdown */
export interface GeneratedSceneResult {
  scene_id: string;
  title: string;
  scene_number: number;
}
```

**Create `frontend/src/components/ScreenplayEditor.tsx`** — a tabbed screenplay editor component.

**Props:**
```typescript
interface ScreenplayEditorProps {
  productionId: string;
}
```

**State:**
- `screenplay: ScreenplayResponse | null` — loaded from API
- `activeTab: string` — current tab: "logline" | "treatment" | "characters" | "breakdown" | "script" | "shotlist"
- `loading: boolean` — initial load
- `regenerating: string | null` — which step is currently regenerating (for spinner)
- `saving: boolean` — save in progress
- `error: string | null`

**On mount:** Call `getScreenplay(productionId)` to load (or auto-create) the screenplay. Set state.

**Tab layout:** 6 horizontal tabs styled with Tailwind (like the department tabs in ProductionBibleDetail). Tabs match SCRN-04 requirements (Logline, Treatment, Scene Breakdown, Script, Shot List) plus Character Breakdowns as a bonus tab:
1. **Logline** — Shows `screenplay.logline` in an editable textarea (2-3 rows). Shows `screenplay.genre` as a secondary field (editable input). Regenerate button calls `generateScreenplayStep(productionId, "logline")`.
2. **Treatment** — Shows `screenplay.treatment` in a larger textarea (8-12 rows). Regenerate button calls `generateScreenplayStep(productionId, "treatment")`.
3. **Character Breakdowns** — If `screenplay.character_breakdowns` is non-null, render a list of CharacterBreakdownEntry cards showing name, role, arc, description, and manifest_tag badge. Editable as raw JSON textarea is acceptable for v1 (structured editing can come later). Regenerate button calls `generateScreenplayStep(productionId, "character-breakdowns")`.
4. **Scene Breakdown** — If `screenplay.scene_breakdown` is non-null, render a numbered list of SceneBreakdownEntry items: scene_number, slugline (bold), intent, emotional_beat, characters_present as comma-joined tags, set_ref, props_required. Each entry in a card-style layout. Regenerate button calls `generateScreenplayStep(productionId, "scene-breakdown")`.
5. **Script** — Shows `screenplay.script` in a large monospace textarea (16-24 rows). Regenerate button calls `generateScreenplayStep(productionId, "script")`.
6. **Shot List** — If `screenplay.shot_list` is non-null, render a grouped list of shot entries organized by scene_number. Each entry shows: shot_number, shot_type (e.g. "WIDE", "CLOSE-UP"), camera_movement, action_notes, and duration_hint. Group shots under scene headings (e.g. "Scene 1", "Scene 2"). If shot_list is null/empty, show placeholder text "Generate Scene Breakdown first, then generate Shot List." Regenerate button calls `generateScreenplayStep(productionId, "shot-list")`.

**Status bar** (at the top of the editor):
- Show current status badge: DRAFT (gray), IN_REVIEW (yellow), LOCKED (green with lock icon).
- Status transition buttons:
  - When DRAFT: "Mark for Review" button → updateScreenplayStatus(productionId, "IN_REVIEW")
  - When IN_REVIEW: "Lock Screenplay" button → updateScreenplayStatus(productionId, "LOCKED"), and "Back to Draft" → updateScreenplayStatus(productionId, "DRAFT")
  - When LOCKED: "Unlock" button → updateScreenplayStatus(productionId, "DRAFT")

**Generate All button** (at the top): "Generate Full Screenplay" button calls `generateScreenplayFull(productionId)`. After calling, start polling `getScreenplay()` every 3 seconds to show incremental progress (check `generating_step` field). When `generating_step` becomes null, stop polling and refresh.

**Generate Scenes button** (only visible when screenplay.status === "LOCKED" AND screenplay.scene_breakdown is non-empty):
- "Generate Scenes from Screenplay" button calls `generateScenesFromScreenplay(productionId)`.
- On success, show a count message: "Created N scenes" and optionally trigger a reload in the parent (ProductionDetail).

**LOCKED behavior:** When `screenplay.status === "LOCKED"`:
- All textareas become `readOnly`
- All Regenerate buttons are disabled (grayed out, `opacity-50 cursor-not-allowed`)
- "Generate Scenes" button remains enabled (scenes are created from locked screenplay)

**Save behavior:** For text fields (logline, treatment, script), debounce saves. On blur or after 2 seconds of inactivity, call `updateScreenplay(productionId, { field: value })`. For structured fields (character_breakdowns, scene_breakdown), save on explicit "Save" button click. Show a subtle save indicator.

**Regeneration UX:** When a Regenerate button is clicked:
1. Set `regenerating = step_name`
2. Call the generation endpoint
3. On response, update `screenplay` state with the new data
4. Clear `regenerating`
5. Show error if request fails

**Styling:** Use Tailwind classes consistent with existing components. White/gray background cards, blue accent for active tab, rounded corners, padding p-4/p-6.

**Option B (Section approach):** If ProductionDetail is a single-scroll layout, add a collapsible "Screenplay" section between the production header and the scene list. Use a disclosure/accordion pattern: click to expand, render `<ScreenplayEditor productionId={productionId} />` inside.

Import `ScreenplayEditor` from `./ScreenplayEditor.tsx`. Add it to the ProductionDetail's JSX.
